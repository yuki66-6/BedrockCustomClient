cmake_minimum_required(VERSION 3.24)
project(BedrockNative LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 and MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
add_library(BedrockNative SHARED src/BedrockNative.cpp)
target_include_directories(BedrockNative PRIVATE include)
target_compile_definitions(BedrockNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(BedrockLauncher WIN32 src/BedrockLauncher.cpp)
target_compile_definitions(BedrockLauncher PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(BedrockLauncher PRIVATE shell32 user32)
